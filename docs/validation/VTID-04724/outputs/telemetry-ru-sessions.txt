-- Which rung spoke, and in which language, per ru session:
with ru as (
  select distinct metadata->>'session_id' as sess
  from oasis_events
  where created_at > now() - interval '10 days'
    and topic = 'vtid.live.session.start'
    and metadata->>'lang' like 'ru%'
)
select e.created_at, right(e.metadata->>'session_id',12) as sess,
       e.metadata->>'stage' as stage,
       left(e.metadata->>'output_preview',240) as output_preview
from oasis_events e join ru on ru.sess = e.metadata->>'session_id'
where e.topic='orb.live.diag' and e.metadata->>'stage'='turn_complete'
order by e.created_at desc;

-- joined against each session's wake_opener:
select right(metadata->>'session_id',12) as sess,
       metadata->>'wake_opener' as wake_opener,
       metadata->>'prompt_len' as plen
from oasis_events
where topic='orb.live.diag' and metadata->>'stage'='greeting_sent'
  and metadata->>'lang' like 'ru%' and created_at > now() - interval '10 days';
